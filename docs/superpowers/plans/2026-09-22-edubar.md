## Interfaces (EduBarCore)

```swift
public struct Course: Identifiable, Hashable, Sendable {
    public let id: String, title: String, start: Date, end: Date, room: String?
    public var shortTitle: String            // sans "T1 - "
}
public enum ICSParser { public static func parse(_ text: String) -> [Course] } // trié par start
public enum Status: Equatable, Sendable {
    case inClass(current: Course, next: Course?, blockEnd: Date, breakFollows: Bool)
    case onBreak(previous: Course, next: Course)
    case beforeFirst(next: Course)
    case dayOver(next: Course?)
    case noClassToday(next: Course?)
}
public struct Schedule: Sendable {
    public init(courses: [Course])
    public func status(at now: Date, calendar: Calendar) -> Status
    public func courses(on day: Date, calendar: Calendar) -> [Course]
}
public struct RoomAlert: Equatable, Sendable { public let from: Course; public let to: Course; public let room: String }
public enum RoomChange { public static func alert(for: Status, at: Date, lead: TimeInterval = 900) -> RoomAlert? }
public enum Display {
    public static func duration(_ seconds: TimeInterval) -> String   // "23 min", "1 h 05"
    public static func time(_ date: Date, calendar: Calendar) -> String   // "8h45", "14h"
    public static func shortRoom(_ room: String) -> String           // "Quai 12 - 501" -> "501"
    public static func dayLabel(_ date: Date, now: Date, calendar: Calendar) -> String
    public static func barText(status: Status, alert: RoomAlert?, now: Date, calendar: Calendar) -> String
}
public enum FeedURL {
    public static func normalize(_ raw: String) -> URL?
    public static func redacted(_ url: URL) -> String
}
```
